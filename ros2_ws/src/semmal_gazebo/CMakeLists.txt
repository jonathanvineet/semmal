cmake_minimum_required(VERSION 3.8)
project(semmal_gazebo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
add_executable(clock_relay src/clock_relay.cpp)
target_compile_features(clock_relay PUBLIC cxx_std_17)
target_link_libraries(clock_relay rclcpp::rclcpp ${rosgraph_msgs_TARGETS})
install(TARGETS clock_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY worlds config scripts media DESTINATION share/${PROJECT_NAME})
ament_package()
